GraphQL Queries for Testing

Create Room (Admin only):

mutation {
  createChatRoom(createRoomInput: {
    name: "General Chat"
    description: "Main discussion room"
  }) {
    id
    name
    description
  }
}

Get All Rooms:

query {
  chatRooms {
    id
    name
    description
    createdAt
  }
}

Get Room Messages:

query {
  chatRoomMessages(roomId: 1) {
    id
    message
    user {
      name
      email
    }
    createdAt
  }
}
